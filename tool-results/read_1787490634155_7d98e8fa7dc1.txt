   185→// ─── Image fallback strategy ───
   186→// Production DB products often have either:
   187→//   • image: null (seed not re-run)
   188→//   • image: '/images/shop/product-XXXX.jpeg' pointing to a file that does NOT exist
   189→//     (uploaded via admin panel, but the upload failed silently or was lost)
   190→//   • slug variants like 'pack-5-stickers-' (trailing dash) or 'packs-10-stickers'
   191→//     that don't match SHOP_PACKS_FALLBACK slugs exactly.
   192→//
   193→// To ALWAYS show a sachet image we:
   194→//   1. Map quantity → local pack image (most reliable, the sachet shows the right number)
   195→//   2. Map slug → local pack image (handles edge cases where quantity differs)
   196→//   3. On <img> onError, swap the broken src to the quantity-mapped local image
   197→//      (instead of hiding it and showing the number fallback, which is less pretty)
   198→const PACK_IMAGE_BY_QUANTITY: Record<number, string> = Object.fromEntries(
   199→  SHOP_PACKS_FALLBACK.map(p => [p.quantity, p.image])
   200→);
   201→const PACK_IMAGE_BY_SLUG: Record<string, string> = Object.fromEntries(
   202→  SHOP_PACKS_FALLBACK.map(p => [p.slug, p.image])
   203→);
   204→// Generic sachet used when quantity is unknown (e.g. Pack Revendeur qty=35)
   205→const GENERIC_PACK_IMAGE = '/images/shop/pack-15-stickers.png';
   206→
   207→/**
   208→ * Resolve the best local image for a given product.
   209→ *
   210→ * CRITICAL: we NEVER trust DB-uploaded images (pattern: /images/shop/product-*.jpeg)
   211→ * because they are almost always broken uploads (admin panel saved the path but
   212→ * the file was never written, or was lost during deploy). We only trust the 4
   213→ * known-good local sachet images that ship with the codebase.
   214→ *
   215→ * Strategy (in order):
   216→ *   1. If DB image matches the known-good local pattern → use it as-is.
   217→ *   2. Otherwise resolve by quantity (most reliable — sachet shows the right number).
   218→ *   3. Then by slug, slug-without-trailing-dash, slug-without-s.
   219→ *   4. Finally fall back to the generic pack-15 sachet.
   220→ *
   221→ * This guarantees the <img src> is ALWAYS a valid local path from the first render,
   222→ * so there is no flash of broken image, no 30s-then-disappear bug, no need to rely
   223→ * on the onError swap (which can race with React re-renders).
   224→ */
   225→function resolvePackImage(p: { image: string | null; quantity: number; slug: string }): string {
   226→  // Only trust DB image if it matches one of our 4 known-good local sachet paths
   227→  if (p.image && /^\/images\/shop\/pack-(3|5|10|15)-stickers\.png$/.test(p.image)) {
   228→    return p.image;
   229→  }
   230→  return (
   231→    PACK_IMAGE_BY_QUANTITY[p.quantity] ||
   232→    PACK_IMAGE_BY_SLUG[p.slug] ||
   233→    PACK_IMAGE_BY_SLUG[p.slug.replace(/[-\s]+$/, '')] || // strip trailing dash: 'pack-5-stickers-' → 'pack-5-stickers'
   234→    PACK_IMAGE_BY_SLUG[p.slug.replace(/^packs-/, 'pack-')] || // 'packs-10-stickers' → 'pack-10-stickers'
   235→    GENERIC_PACK_IMAGE
   236→  );
   237→}
   238→
   239→/**
   240→ * Decide what to use as the <img src>. Returns the DB image ONLY if it's safe
   241→ * (one of the 4 known-good local paths). Otherwise returns the fallback.
   242→ *
   243→ * This prevents the "30-second disappear" bug where:
   244→ *   1. Initial render uses SHOP_PACKS_FALLBACK (sachets show).
   245→ *   2. API responds ~30s later with broken DB image URLs.
   246→ *   3. <img src> switches to broken URL → broken image icon.
   247→ *   4. onError swap may race with React re-renders and lose the swap.
   248→ *
   249→ * By NEVER using a DB image that doesn't match the safe pattern, we eliminate
   250→ * the broken-image state entirely.
   251→ */
   252→function getSafeImageSrc(p: { image: string | null; fallbackImage: string }): string {
   253→  if (p.image && /^\/images\/shop\/pack-(3|5|10|15)-stickers\.png$/.test(p.image)) {
   254→    return p.image;
   255→  }
   256→  return p.fallbackImage;
   257→}
