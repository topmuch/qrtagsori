    27→    27→import { Badge } from "@/components/ui/badge";

    29→    29→import { PRODUCT_TYPES } from '@/lib/agency-types';
    30→    30→
    31→    31→// Types
    32→    32→interface Agency {
    33→    33→  id: string;
    34→    34→  name: string;
    35→    35→  slug: string;
    36→    36→  email: string | null;
    37→    37→  phone: string | null;
    38→    38→  active: boolean;
    39→    39→  createdAt: string;
    40→    40→}
    41→    41→
    42→    42→type GenerationContext = 'individual' | 'agency';
    43→    43→
    44→    44→export default function GenererQRPage() {
    45→    45→  const [agencies, setAgencies] = useState<Agency[]>([]);
    46→    46→  const [loading, setLoading] = useState(false);
    47→    47→  const [qrGenerating, setQrGenerating] = useState(false);
    48→    48→  const [successMessage, setSuccessMessage] = useState('');
    49→    49→  const [errorMessage, setErrorMessage] = useState('');
    50→    50→  const [lastGeneratedRefs, setLastGeneratedRefs] = useState<string[]>([]);
    51→    51→  const [isExporting, setIsExporting] = useState(false);
    52→    52→
    53→    53→  // Context selection
    54→    54→  const [context, setContext] = useState<GenerationContext>('agency');
    55→    55→
    56→    56→  // Individual form — QRTags : duration forcée à '1y', productType remplace baggageCount
    57→    57→  const [individualForm, setIndividualForm] = useState({
    58→    58→    firstName: '',
    59→    59→    lastName: '',
    60→    60→    whatsapp: '',
    61→    61→    duration: '1y' as '1y', // QRTags : 1 an uniquement (7 jours supprimé)
    62→    62→    productType: 'laptop',  // QRTags : type d'objet à protéger
    63→    63→  });
    64→    64→
    65→    65→  // Agency form — QRTags : productType remplace baggagePerTraveler
    66→    66→  const [agencyForm, setAgencyForm] = useState({
    67→    67→    type: 'qrtags' as 'hajj' | 'qrtags',
    68→    68→    agencyId: '',
    69→    69→    travelerCount: 1,
    70→    70→    productType: 'laptop', // QRTags : type d'objet à protéger
    71→    71→  });
    72→    72→
    73→    73→  useEffect(() => {
    74→    74→    fetchAgencies();
    75→    75→  }, []);
    76→    76→
    77→    77→  const fetchAgencies = async () => {
    78→    78→    setLoading(true);
    79→    79→    try {
    80→    80→      const res = await fetch('/api/admin/agencies');
    81→    81→      const data = await res.json();
    82→    82→      setAgencies(data.agencies || []);
    83→    83→    } catch (error) {
    84→    84→      console.error('Error fetching agencies:', error);
    85→    85→    } finally {
    86→    86→      setLoading(false);
    87→    87→    }
    88→    88→  };
    89→    89→
    90→    90→  // Calculate QR count for display
    91→    91→  // QRTags : 1 tag par objet (plus de notion de "baggageCount" ou "baggagePerTraveler")
    92→    92→  const getQrCount = () => {
    93→    93→    if (context === 'individual') {
    94→    94→      return 1; // 1 tag = 1 objet
    95→    95→    }
    96→    96→    return agencyForm.travelerCount; // 1 tag par client
    97→    97→  };
    98→    98→
    99→    99→  // Validate individual form
   100→   100→  const validateIndividualForm = (): boolean => {
   101→   101→    if (!individualForm.firstName.trim()) {
   102→   102→      setErrorMessage('Le prénom est requis');
   103→   103→      return false;
   104→   104→    }
   105→   105→    if (!individualForm.lastName.trim()) {
   106→   106→      setErrorMessage('Le nom est requis');
   107→   107→      return false;
   108→   108→    }
   109→   109→    if (!individualForm.whatsapp.trim()) {
   110→   110→      setErrorMessage('Le numéro WhatsApp est requis');
   111→   111→      return false;
   112→   112→    }
   113→   113→    // Basic WhatsApp format validation
   114→   114→    const phoneRegex = /^\+?[1-9]\d{6,14}$/;
   115→   115→    if (!phoneRegex.test(individualForm.whatsapp.replace(/\s/g, ''))) {
   116→   116→      setErrorMessage('Format WhatsApp invalide (ex: +33612345678)');
   117→   117→      return false;
   118→   118→    }
   119→   119→    return true;
   120→   120→  };
   121→   121→
   122→   122→  // Validate agency form
   123→   123→  const validateAgencyForm = (): boolean => {
   124→   124→    if (!agencyForm.agencyId) {
   125→   125→      setErrorMessage('Veuillez sélectionner une agence');
   126→   126→      return false;
   127→   127→    }
   128→   128→    return true;
   129→   129→  };
   130→   130→
   131→   131→  // Export generated QR codes as PNG images (one ZIP-like download per QR)
   132→   132→  const handleExportGenerated = async () => {
   133→   133→    if (lastGeneratedRefs.length === 0) return;
   134→   134→    setIsExporting(true);
   135→   135→    try {
   136→   136→      const QRCode = (await import('qrcode')).default;
   137→   137→      const baseUrl = window.location.origin;
   138→   138→
   139→   139→      // Générer chaque QR code en PNG et télécharger
   140→   140→      for (let i = 0; i < lastGeneratedRefs.length; i++) {
   141→   141→        const ref = lastGeneratedRefs[i];
   142→   142→        const targetUrl = `${baseUrl}/scan/${ref}`;
   143→   143→
   144→   144→        // Générer le QR code en PNG (base64)
   145→   145→        const dataUrl = await QRCode.toDataURL(targetUrl, {
   146→   146→          errorCorrectionLevel: 'H',
   147→   147→          margin: 2,
   148→   148→          width: 512,
   149→   149→          color: {
   150→   150→            dark: '#111111',
   151→   151→            light: '#E3B23C',
   152→   152→          },
   153→   153→        });
   154→   154→
   155→   155→        // Convertir base64 en blob et télécharger
   156→   156→        const response = await fetch(dataUrl);
   157→   157→        const blob = await response.blob();
   158→   158→        const url = URL.createObjectURL(blob);
   159→   159→        const a = document.createElement('a');
   160→   160→        a.href = url;
   161→   161→        a.download = `QRTags-${ref}.png`;
   162→   162→        document.body.appendChild(a);
   163→   163→        a.click();
   164→   164→        document.body.removeChild(a);
   165→   165→        URL.revokeObjectURL(url);
   166→   166→
   167→   167→        // Petite pause entre chaque téléchargement
   168→   168→        await new Promise(resolve => setTimeout(resolve, 300));
   169→   169→      }
   170→   170→
   171→   171→      setSuccessMessage(`${lastGeneratedRefs.length} QR codes exportés en PNG`);
   172→   172→      setTimeout(() => setSuccessMessage(''), 5000);
   173→   173→    } catch (error) {
   174→   174→      console.error('Export error:', error);
   175→   175→      setErrorMessage('Erreur lors de l\'export des QR codes');
   176→   176→    } finally {
   177→   177→      setIsExporting(false);
   178→   178→    }
   179→   179→  };
   180→   180→  const handleGenerateQR = async () => {
   181→   181→    setErrorMessage('');
   182→   182→    
   183→   183→    // Validate based on context
   184→   184→    if (context === 'individual' && !validateIndividualForm()) {
   185→   185→      return;
   186→   186→    }
   187→   187→    if (context === 'agency' && !validateAgencyForm()) {
   188→   188→      return;
   189→   189→    }
   190→   190→    
   191→   191→    setQrGenerating(true);
   192→   192→    
   193→   193→    try {
   194→   194→      const payload = context === 'individual' 
   195→   195→        ? {
   196→   196→            context: 'individual',
   197→   197→            type: 'qrtags' as const,
   198→   198→            firstName: individualForm.firstName.trim(),
   199→   199→            lastName: individualForm.lastName.trim(),
   200→   200→            whatsapp: individualForm.whatsapp.trim(),
   201→   201→            duration: '1y', // QRTags : 1 an uniquement
   202→   202→            baggageCount: 1, // 1 tag = 1 objet
   203→   203→          }
   204→   204→        : {
   205→   205→            context: 'agency',
   206→   206→            type: agencyForm.type,
   207→   207→            agencyId: agencyForm.agencyId,
   208→   208→            travelerCount: agencyForm.travelerCount,
   209→   209→            count: 1, // QRTags : 1 tag par client (1 objet)
   210→   210→          };
   211→   211→      
   212→   212→      const response = await fetch('/api/admin/baggages/generate', {
   213→   213→        method: 'POST',
   214→   214→        headers: { 'Content-Type': 'application/json' },
   215→   215→        body: JSON.stringify(payload),
   216→   216→      });
   217→   217→      
   218→   218→      const data = await response.json();
   219→   219→      
   220→   220→      if (response.ok) {
   221→   221→        setSuccessMessage(`${data.generated} tag(s) QRTags généré(s) avec succès !`);
   222→   222→        setLastGeneratedRefs(data.references || []);
   223→   223→        // Reset individual form
   224→   224→        if (context === 'individual') {
   225→   225→          setIndividualForm({
   226→   226→            firstName: '',
   227→   227→            lastName: '',
   228→   228→            whatsapp: '',
   229→   229→            duration: '1y', // QRTags : 1 an par défaut
   230→   230→            productType: 'laptop',
   231→   231→          });
   232→   232→        }
   233→   233→        setTimeout(() => {
   234→   234→          setSuccessMessage('');
   235→   235→          setLastGeneratedRefs([]);
   236→   236→        }, 10000);
   237→   237→      } else {
   238→   238→        setErrorMessage(data.error || 'Erreur lors de la génération');
   239→   239→      }
   240→   240→    } catch (error) {
   241→   241→      console.error('Error generating QR codes:', error);
   242→   242→      setErrorMessage('Erreur lors de la génération des QR codes');
   243→   243→    } finally {
   244→   244→      setQrGenerating(false);
   245→   245→    }
   246→   246→  };
   247→   247→
   248→   248→  return (
   249→   249→    <div className="max-w-4xl mx-auto">
   250→   250→      {/* Page Header */}
   251→   251→      <div className="mb-8">
   252→   252→        <h1 className="text-2xl font-bold text-slate-800 dark:text-white">Génération QRTags</h1>
   253→   253→        <p className="text-slate-500 dark:text-slate-400 mt-1">Générez des tags QRTags pour retrouver vos objets perdus</p>
   254→   254→      </div>
   255→   255→
   256→   256→      {/* Success Message */}
   257→   257→      {successMessage && (
   258→   258→        <div className="mb-6 bg-emerald-50 dark:bg-emerald-600/10 border border-emerald-200 dark:border-emerald-800 text-emerald-700 dark:text-emerald-400 px-4 py-4 rounded-xl">
   259→   259→          <div className="flex items-center gap-2 mb-3">
   260→   260→            <CheckCircle className="w-5 h-5" />
   261→   261→            <span className="font-medium">{successMessage}</span>
   262→   262→          </div>
   263→   263→          {lastGeneratedRefs.length > 0 && (
   264→   264→            <div className="flex flex-wrap gap-2 mt-3">
   265→   265→              <button
   266→   266→                onClick={handleExportGenerated}
   267→   267→                disabled={isExporting}
   268→   268→                className="inline-flex items-center gap-2 px-4 py-2 bg-gradient-to-r from-[#1e7e34] to-[#0d5e34] text-white rounded-lg hover:from-[#228b22] hover:to-[#1e7e34] transition-all text-sm shadow-lg shadow-green-900/20 disabled:opacity-50"
   269→   269→              >
   270→   270→                {isExporting ? (
   271→   271→                  <>
   272→   272→                    <Loader2 className="w-4 h-4 animate-spin" />
   273→   273→                    Export en cours...
   274→   274→                  </>
   275→   275→                ) : (
   276→   276→                  <>
   277→   277→                    <Archive className="w-4 h-4" />
   278→   278→                    Exporter en ZIP ({lastGeneratedRefs.length} QR)
   279→   279→                  </>
   280→   280→                )}
   281→   281→              </button>
   282→   282→              <a
   283→   283→                href="/admin/qrcodes"
   284→   284→                className="inline-flex items-center gap-2 px-4 py-2 bg-blue-600 text-white rounded-lg hover:bg-blue-700 transition-colors text-sm"
   285→   285→              >
   286→   286→                <QrCode className="w-4 h-4" />
   287→   287→                Voir tous les QR codes
   288→   288→              </a>
   289→   289→            </div>
   290→   290→          )}
   291→   291→        </div>
   292→   292→      )}
   293→   293→
   294→   294→      {/* Context Selector */}
   295→   295→      <div className="mb-6">
   296→   296→        <Label className="text-slate-700 dark:text-slate-300 mb-3">Mode de génération</Label>
   297→   297→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-3 mt-2">
   298→   298→          <button
   299→   299→            onClick={() => setContext('individual')}
   300→   300→            className={cn(
   301→   301→              "flex items-center gap-3 p-4 rounded-xl border transition-all",
   302→   302→              context === 'individual' 
   303→   303→                ? "bg-[#2563EB] border-[#2563EB] text-white" 
   304→   304→                : "bg-white dark:bg-slate-900 border-slate-200 dark:border-slate-700 text-slate-700 dark:text-slate-300 hover:border-slate-300"
   305→   305→            )}
   306→   306→          >
   307→   307→            <User className="w-5 h-5" />
   308→   308→            <div className="text-left">
   309→   309→              <p className="font-medium">Tag individuel</p>
   310→   310→              <p className="text-xs opacity-80">1 tag, sans entreprise</p>
   311→   311→            </div>
   312→   312→          </button>
   313→   313→          <button
   314→   314→            onClick={() => setContext('agency')}
   315→   315→            className={cn(
   316→   316→              "flex items-center gap-3 p-4 rounded-xl border transition-all",
   317→   317→              context === 'agency' 
   318→   318→                ? "bg-[#2563EB] border-[#2563EB] text-white" 
   319→   319→                : "bg-white dark:bg-slate-900 border-slate-200 dark:border-slate-700 text-slate-700 dark:text-slate-300 hover:border-slate-300"
   320→   320→            )}
   321→   321→          >
   322→   322→            <Building2 className="w-5 h-5" />
   323→   323→            <div className="text-left">
   324→   324→              <p className="font-medium">Agence partenaire</p>
   325→   325→              <p className="text-xs opacity-80">Lot de tags pour entreprise</p>
   326→   326→            </div>
   327→   327→          </button>
   328→   328→        </div>
   329→   329→      </div>
   330→   330→
   331→   331→      <div className="grid lg:grid-cols-2 gap-6">
   332→   332→        {/* Main Form Card */}
   333→   333→        <Card className="bg-white dark:bg-slate-900 border-slate-200 dark:border-slate-800 shadow-sm rounded-2xl">
   334→   334→          <CardHeader>
   335→   335→            <CardTitle className="text-slate-800 dark:text-white flex items-center gap-2">
   336→   336→              <QrCode className="w-5 h-5 text-blue-600" />
   337→   337→              {context === 'individual' ? 'Tag individuel' : 'Lot pour entreprise'}
   338→   338→            </CardTitle>
   339→   339→          </CardHeader>
   340→   340→          <CardContent className="space-y-4">
   341→   341→            {/* Individual Form */}
   342→   342→            {context === 'individual' ? (
   343→   343→              <>
   344→   344→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   345→   345→                  <div className="space-y-2">
   346→   346→                    <Label className="text-slate-700 dark:text-slate-300">Prénom *</Label>
   347→   347→                    <Input 
   348→   348→                      value={individualForm.firstName}
   349→   349→                      onChange={(e) => setIndividualForm({ ...individualForm, firstName: e.target.value })}
   350→   350→                      placeholder="Ahmed"
   351→   351→                      className="bg-white dark:bg-slate-800 border-slate-200 dark:border-slate-700 text-slate-800 dark:text-white"
   352→   352→                    />
   353→   353→                  </div>
   354→   354→                  <div className="space-y-2">
   355→   355→                    <Label className="text-slate-700 dark:text-slate-300">Nom *</Label>
   356→   356→                    <Input 
   357→   357→                      value={individualForm.lastName}
   358→   358→                      onChange={(e) => setIndividualForm({ ...individualForm, lastName: e.target.value })}
   359→   359→                      placeholder="Diop"
   360→   360→                      className="bg-white dark:bg-slate-800 border-slate-200 dark:border-slate-700 text-slate-800 dark:text-white"
   361→   361→                    />
   362→   362→                  </div>
   363→   363→                </div>
   364→   364→
   365→   365→                <div className="space-y-2">
   366→   366→                  <Label className="text-slate-700 dark:text-slate-300">WhatsApp *</Label>
   367→   367→                  <Input 
   368→   368→                    value={individualForm.whatsapp}
   369→   369→                    onChange={(e) => setIndividualForm({ ...individualForm, whatsapp: e.target.value })}
   370→   370→                    placeholder="+33612345678"
   371→   371→                    className="bg-white dark:bg-slate-800 border-slate-200 dark:border-slate-700 text-slate-800 dark:text-white"
   372→   372→                  />
   373→   373→                </div>
   374→   374→
   375→   375→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   376→   376→                  <div className="space-y-2">

   379→   379→                      <SelectTrigger className="bg-white dark:bg-slate-800 border-slate-200 dark:border-slate-700 text-slate-800 dark:text-white">
   380→   380→                        <SelectValue />
   381→   381→                      </SelectTrigger>
   382→   382→                      <SelectContent className="bg-white dark:bg-slate-900 border-slate-200 dark:border-slate-800">
   383→   383→                        <SelectItem value="1y">1 an</SelectItem>
   384→   384→                      </SelectContent>
   385→   385→                    </Select>
   386→   386→                  </div>
   387→   387→                  <div className="space-y-2">
   388→   388→                    <Label className="text-slate-700 dark:text-slate-300">Objet à protéger</Label>
   389→   389→                    <Select

   394→   394→                        <SelectValue />
   395→   395→                      </SelectTrigger>
   396→   396→                      <SelectContent className="bg-white dark:bg-slate-900 border-slate-200 dark:border-slate-800 max-h-72 overflow-y-auto">
   397→   397→                        {PRODUCT_TYPES.map((p) => (
   398→   398→                          <SelectItem key={p.value} value={p.value}>
   399→   399→                            {p.icon} {p.label}
   400→   400→                          </SelectItem>
   401→   401→                        ))}
   402→   402→                      </SelectContent>
   403→   403→                    </Select>
   404→   404→                  </div>
   405→   405→                </div>
   406→   406→
   407→   407→                <div className="bg-amber-50 dark:bg-blue-600/10 border border-amber-200 dark:border-amber-800 rounded-xl p-4 text-sm text-amber-700 dark:text-blue-500">

   409→   409→                </div>
   410→   410→
   411→   411→                <div className="bg-slate-50 dark:bg-slate-800 rounded-xl p-4 text-sm text-slate-600 dark:text-slate-300">

   426→   426→                      <SelectValue />
   427→   427→                    </SelectTrigger>
   428→   428→                    <SelectContent className="bg-white dark:bg-slate-900 border-slate-200 dark:border-slate-800">

   430→   430→                    </SelectContent>
   431→   431→                  </Select>
   432→   432→                </div>
   433→   433→
   434→   434→                <div className="space-y-2">
   435→   435→                  <Label className="text-slate-700 dark:text-slate-300">Agence partenaire *</Label>

   439→   439→                  >
