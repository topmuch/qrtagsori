    28→interface DisplayBaggage {

    41→
    42→function toDisplay(b: LocalBaggageItem | TravelerBaggage): DisplayBaggage {
    43→  if ('objectInfo' in b) {
    44→    const lb = b as LocalBaggageItem;
    45→    return {
    46→      reference: lb.reference,
    47→      status: lb.status,
    48→      travelerName: lb.travelerName,
    49→      lastScanDate: lb.lastScanDate,
    50→      lastScanLocation: lb.lastScanLocation,
    51→      scanCount: lb.scanCount,
    52→      trackingToken: lb.trackingToken,
    53→      expiresAt: lb.expiresAt,
    54→      objectName: lb.objectInfo?.object_name,
    55→      color: lb.objectInfo?.color,
    56→      categoryLabel: lb.objectInfo?.category_label,
    57→    };
    58→  }
    59→  const tb = b as TravelerBaggage;
    60→  return {
    61→    reference: tb.reference,
    62→    status: tb.status,
    63→    travelerName: '',
    64→    lastScanDate: tb.lastScanDate,
    65→    lastScanLocation: tb.lastScanLocation,
    66→    scanCount: tb.scanCount,
    67→    trackingToken: tb.trackingToken,
    68→    expiresAt: tb.expiresAt,
    69→    objectName: tb.customData?.object_name,
    70→    color: tb.customData?.color,
    71→    categoryLabel: tb.customData?.category_label,
    72→  };
    73→}

   572→          {!showReviewForm ? (
   573→            <button
   574→              onClick={() => { setShowReviewForm(true); setReviewName(traveler?.name || ''); }}

   576→            >

   579→                <p className="text-sm font-bold text-[#1a1a1a]">Laisser un avis ⭐</p>
   580→                <p className="text-xs text-[#525252]">Partagez votre expérience avec QRTags</p>

   582→              <Plus className="w-5 h-5 text-[#E3B23C]" />

   584→          ) : (
   585→            <div className="rounded-2xl border-2 border-[#E3B23C]/40 bg-[#FFFDF5] overflow-hidden">
   586→              <div className="p-4 flex items-center justify-between border-b border-[#E3B23C]/20">
   587→                <div className="flex items-center gap-2">
   588→                  <Star className="w-5 h-5 text-[#E3B23C]" />
   589→                  <span className="text-sm font-bold text-[#1a1a1a]">Votre avis</span>
   590→                </div>
   591→                <button onClick={() => setShowReviewForm(false)} className="text-[#525252] hover:text-[#1a1a1a]">
   592→                  <X className="w-5 h-5" />
   593→                </button>
   594→              </div>
   595→              <div className="p-4 space-y-4">

   597→                <div>
   598→                  <label className="block text-xs font-bold text-[#525252] mb-1.5">Votre nom</label>

   600→                    type="text"

   604→                    className="w-full px-4 py-2.5 rounded-xl bg-white border border-[#e5e5e5] text-[#1a1a1a] text-sm placeholder:text-[#a3a3a3] focus:ring-2 focus:ring-[#E3B23C] focus:border-transparent"

   613→                      value={selectedBaggageRef}
   614→                      onChange={(e) => setSelectedBaggageRef(e.target.value)}

   616→                    >
   617→                      <option value="">— Aucun objet en particulier —</option>
   618→                      {displayBaggages.map(b => (
   619→                        <option key={b.reference} value={b.reference}>
   620→                          {b.objectName ? `${b.objectName} (${b.reference})` : b.reference}
   621→                        </option>
   622→                      ))}
   623→                    </select>
   624→                  </div>
   625→                )}
   626→
   627→                {/* Étoiles */}
   628→                <div>
   629→                  <label className="block text-xs font-bold text-[#525252] mb-1.5">Votre note</label>
   630→                  <div className="flex items-center gap-1">
   631→                    {[1, 2, 3, 4, 5].map((star) => (
   632→                      <button
   633→                        key={star}
   634→                        type="button"
   635→                        onClick={() => setReviewRating(star)}
   636→                        className="transition-transform hover:scale-110"
   637→                      >
   638→                        <Star
   639→                          width={28}
   640→                          height={28}
   641→                          className={star <= reviewRating ? 'fill-current text-[#E3B23C]' : 'text-[#d4d4d4]'}
   642→                        />
   643→                      </button>
   644→                    ))}
   645→                    <span className="ml-2 text-sm font-bold text-[#1a1a1a]">{reviewRating}/5</span>
   646→                  </div>
   647→                </div>
   648→
   649→                {/* Commentaire */}
   650→                <div>
   651→                  <label className="block text-xs font-bold text-[#525252] mb-1.5">Votre commentaire</label>
   652→                  <textarea
   653→                    value={reviewContent}
   654→                    onChange={(e) => setReviewContent(e.target.value)}
   655→                    placeholder="Partagez votre expérience..."
   656→                    rows={3}
   657→                    className="w-full px-4 py-2.5 rounded-xl bg-white border border-[#e5e5e5] text-[#1a1a1a] text-sm placeholder:text-[#a3a3a3] focus:ring-2 focus:ring-[#E3B23C] focus:border-transparent resize-none"
   658→                  />
   659→                </div>
   660→
   661→                {/* Erreur */}
   662→                {reviewError && (
   663→                  <p className="text-xs text-red-600 font-medium bg-red-50 rounded-lg p-2.5">{reviewError}</p>
   664→                )}
   665→
   666→                {/* Succès */}
   667→                {reviewSuccess && (
   668→                  <p className="text-xs text-green-700 font-medium bg-green-50 rounded-lg p-2.5">✅ Merci pour votre avis ! Il sera publié après validation.</p>
   669→                )}
   670→
   671→                {/* Bouton */}
   672→                <button
   673→                  onClick={handleSubmitReview}
   674→                  disabled={reviewSubmitting || reviewContent.trim().length < 10 || reviewName.trim().length < 2}
   675→                  className="w-full py-3 bg-[#E3B23C] text-black text-sm font-bold rounded-xl hover:bg-[#E3B23C]/80 transition disabled:opacity-50 flex items-center justify-center gap-2"
   676→                >
   677→                  {reviewSubmitting ? (
   678→                    <Loader2 className="w-4 h-4 animate-spin" />
   679→                  ) : null}
   680→                  {reviewSubmitting ? 'Envoi en cours...' : 'Envoyer mon avis'}
   681→                </button>
   682→              </div>
   683→            </div>
   684→          )}
   685→        </div>
   686→      )}
   687→    </main>
   688→  );
   689→}

   691→// Helper: convert base64 VAPID key to Uint8Array
   692→function urlBase64ToUint8Array(base64String: string): Uint8Array {
   693→  const padding = '='.repeat((4 - base64String.length % 4) % 4);
   694→  const base64 = (base64String + padding).replace(/\-/g, '+').replace(/_/g, '/');
   695→  const rawData = window.atob(base64);
   696→  const outputArray = new Uint8Array(rawData.length);
   697→  for (let i = 0; i < rawData.length; ++i) {
   698→    outputArray[i] = rawData.charCodeAt(i);
   699→  }
   700→  return outputArray;
   701→}
