   691→   691→// Helper: convert base64 VAPID key to Uint8Array
   692→   692→function urlBase64ToUint8Array(base64String: string): Uint8Array {
   693→   693→  const padding = '='.repeat((4 - base64String.length % 4) % 4);
   694→   694→  const base64 = (base64String + padding).replace(/\-/g, '+').replace(/_/g, '/');
   695→   695→  const rawData = window.atob(base64);
   696→   696→  const outputArray = new Uint8Array(rawData.length);
   697→   697→  for (let i = 0; i < rawData.length; ++i) {
   698→   698→    outputArray[i] = rawData.charCodeAt(i);
   699→   699→  }
   700→   700→  return outputArray;
   701→   701→}
