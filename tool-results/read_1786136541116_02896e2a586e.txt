   180→// ─── Image fallback strategy ───
   181→// Production DB products often have either:
   182→//   • image: null (seed not re-run)
   183→//   • image: '/images/shop/product-XXXX.jpeg' pointing to a file that does NOT exist
   184→//     (uploaded via admin panel, but the upload failed silently or was lost)
   185→//   • slug variants like 'pack-5-stickers-' (trailing dash) or 'packs-10-stickers'
   186→//     that don't match SHOP_PACKS_FALLBACK slugs exactly.
   187→//
   188→// To ALWAYS show a sachet image we:
   189→//   1. Map quantity → local pack image (most reliable, the sachet shows the right number)
   190→//   2. Map slug → local pack image (handles edge cases where quantity differs)
   191→//   3. On <img> onError, swap the broken src to the quantity-mapped local image
   192→//      (instead of hiding it and showing the number fallback, which is less pretty)
   193→const PACK_IMAGE_BY_QUANTITY: Record<number, string> = Object.fromEntries(
   194→  SHOP_PACKS_FALLBACK.map(p => [p.quantity, p.image])
   195→);
   196→const PACK_IMAGE_BY_SLUG: Record<string, string> = Object.fromEntries(
   197→  SHOP_PACKS_FALLBACK.map(p => [p.slug, p.image])
   198→);
   199→// Generic sachet used when quantity is unknown (e.g. Pack Revendeur qty=35)
   200→const GENERIC_PACK_IMAGE = '/images/shop/pack-15-stickers.png';
   201→
   202→/**
   203→ * Resolve the best local image for a given product.
   204→ *
   205→ * CRITICAL: we NEVER trust DB-uploaded images (pattern: /images/shop/product-*.jpeg)
   206→ * because they are almost always broken uploads (admin panel saved the path but
   207→ * the file was never written, or was lost during deploy). We only trust the 4
   208→ * known-good local sachet images that ship with the codebase.
   209→ *
   210→ * Strategy (in order):
   211→ *   1. If DB image matches the known-good local pattern → use it as-is.
   212→ *   2. Otherwise resolve by quantity (most reliable — sachet shows the right number).
   213→ *   3. Then by slug, slug-without-trailing-dash, slug-without-s.
   214→ *   4. Finally fall back to the generic pack-15 sachet.
   215→ *
   216→ * This guarantees the <img src> is ALWAYS a valid local path from the first render,
   217→ * so there is no flash of broken image, no 30s-then-disappear bug, no need to rely
   218→ * on the onError swap (which can race with React re-renders).
   219→ */
   220→function resolvePackImage(p: { image: string | null; quantity: number; slug: string }): string {
   221→  // Only trust DB image if it matches one of our 4 known-good local sachet paths
   222→  if (p.image && /^\/images\/shop\/pack-(3|5|10|15)-stickers\.png$/.test(p.image)) {
   223→    return p.image;
   224→  }
   225→  return (
   226→    PACK_IMAGE_BY_QUANTITY[p.quantity] ||
   227→    PACK_IMAGE_BY_SLUG[p.slug] ||
   228→    PACK_IMAGE_BY_SLUG[p.slug.replace(/[-\s]+$/, '')] || // strip trailing dash: 'pack-5-stickers-' → 'pack-5-stickers'
   229→    PACK_IMAGE_BY_SLUG[p.slug.replace(/^packs-/, 'pack-')] || // 'packs-10-stickers' → 'pack-10-stickers'
   230→    GENERIC_PACK_IMAGE
   231→  );
   232→}
   233→
   234→/**
   235→ * Decide what to use as the <img src>. Returns the DB image ONLY if it's safe
   236→ * (one of the 4 known-good local paths). Otherwise returns the fallback.
   237→ *
   238→ * This prevents the "30-second disappear" bug where:
   239→ *   1. Initial render uses SHOP_PACKS_FALLBACK (sachets show).
   240→ *   2. API responds ~30s later with broken DB image URLs.
   241→ *   3. <img src> switches to broken URL → broken image icon.
   242→ *   4. onError swap may race with React re-renders and lose the swap.
   243→ *
   244→ * By NEVER using a DB image that doesn't match the safe pattern, we eliminate
   245→ * the broken-image state entirely.
   246→ */
   247→function getSafeImageSrc(p: { image: string | null; fallbackImage: string }): string {
   248→  if (p.image && /^\/images\/shop\/pack-(3|5|10|15)-stickers\.png$/.test(p.image)) {
   249→    return p.image;
   250→  }
   251→  return p.fallbackImage;
   252→}
