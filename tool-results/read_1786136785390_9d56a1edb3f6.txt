   105→   105→// ═══════════════════════════════════════════════════════════════════════
   106→   106→
   107→   107→function DesignGeneration({ agencies }: { agencies: Agency[] }) {
   108→   108→  // Template state
   109→   109→  const [templateInfo, setTemplateInfo] = useState<TemplateInfo | null>(null);
   110→   110→  const [templateLoading, setTemplateLoading] = useState(true);
   111→   111→  const [uploading, setUploading] = useState(false);
   112→   112→  const fileInputRef = useRef<HTMLInputElement>(null);
   113→   113→
   114→   114→  // Generation state
   115→   115→  const [agencyId, setAgencyId] = useState('');
   116→   116→  const [quantity, setQuantity] = useState(10);
   117→   117→  const [sizeCm, setSizeCm] = useState<StickerSize>(4);
   118→   118→  const [generating, setGenerating] = useState(false);
   119→   119→  const [references, setReferences] = useState<string[]>([]);
   120→   120→
   121→   121→  // Export state
   122→   122→  const [exportingPdf, setExportingPdf] = useState(false);
   123→   123→  const [exportingZip, setExportingZip] = useState(false);
   124→   124→
   125→   125→  // Preview state
   126→   126→  const [previewUrl, setPreviewUrl] = useState<string | null>(null);
   127→   127→  const [previewLoading, setPreviewLoading] = useState(false);
   128→   128→
   129→   129→  // Messages
   130→   130→  const [successMsg, setSuccessMsg] = useState('');
   131→   131→  const [errorMsg, setErrorMsg] = useState('');
   132→   132→
   133→   133→  // Load template info on mount
   134→   134→  useEffect(() => {
   135→   135→    loadTemplateInfo();
   136→   136→  }, []);
   137→   137→
   138→   138→  // Auto-preview when we have a reference and template
   139→   139→  useEffect(() => {
   140→   140→    if (references.length > 0 && templateInfo?.exists) {
   141→   141→      loadPreview(references[0]);
   142→   142→    } else {
   143→   143→      setPreviewUrl(null);
   144→   144→    }
   145→   145→  }, [references, templateInfo?.exists, sizeCm]);
   146→   146→
   147→   147→  const loadTemplateInfo = async () => {
   148→   148→    setTemplateLoading(true);
   149→   149→    try {
   150→   150→      const res = await fetch('/api/admin/qr-design/upload');
   151→   151→      const data = await res.json();
   152→   152→      setTemplateInfo(data.template);
   153→   153→    } catch {
   154→   154→      setTemplateInfo(null);
   155→   155→    } finally {
   156→   156→      setTemplateLoading(false);
   157→   157→    }
   158→   158→  };
   159→   159→
   160→   160→  const handleUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   161→   161→    const file = e.target.files?.[0];
   162→   162→    if (!file) return;
   163→   163→
   164→   164→    setUploading(true);
   165→   165→    setErrorMsg('');
   166→   166→    setSuccessMsg('');
   167→   167→
   168→   168→    const formData = new FormData();
   169→   169→    formData.append('template', file);
   170→   170→
   171→   171→    try {
   172→   172→      const res = await fetch('/api/admin/qr-design/upload', {
   173→   173→        method: 'POST',
   174→   174→        body: formData,
   175→   175→      });
   176→   176→      const data = await res.json();
   177→   177→
   178→   178→      if (res.ok) {
   179→   179→        setSuccessMsg('Template de design enregistré !');
   180→   180→        setTemplateInfo(data.template);
   181→   181→      } else {
   182→   182→        setErrorMsg(data.error || "Erreur lors de l'upload");
   183→   183→      }
   184→   184→    } catch {
   185→   185→      setErrorMsg("Erreur lors de l'upload du template.");
   186→   186→    } finally {
   187→   187→      setUploading(false);
   188→   188→    }
   189→   189→  };
   190→   190→
   191→   191→  const handleGenerate = async () => {
   192→   192→    if (!templateInfo?.exists) {
   193→   193→      setErrorMsg('Veuillez d\'abord uploader un template de design.');
   194→   194→      return;
   195→   195→    }
   196→   196→
   197→   197→    setErrorMsg('');
   198→   198→    setSuccessMsg('');
   199→   199→    setGenerating(true);
   200→   200→
   201→   201→    try {
   202→   202→      const res = await fetch('/api/admin/qr-design/generate', {
   203→   203→        method: 'POST',
   204→   204→        headers: { 'Content-Type': 'application/json' },
   205→   205→        body: JSON.stringify({
   206→   206→          agencyId: agencyId || undefined,
   207→   207→          quantity,
   208→   208→        }),
   209→   209→      });
   210→   210→      const data = await res.json();
   211→   211→
   212→   212→      if (res.ok) {
   213→   213→        setReferences(data.references || []);
   214→   214→        setSuccessMsg(`${data.generated} QR code(s) généré(s) avec succès !`);
   215→   215→      } else {
   216→   216→        setErrorMsg(data.error || 'Erreur lors de la génération.');
   217→   217→      }
   218→   218→    } catch {
   219→   219→      setErrorMsg('Erreur lors de la génération.');
   220→   220→    } finally {
   221→   221→      setGenerating(false);
   222→   222→    }
   223→   223→  };
   224→   224→
   225→   225→  const loadPreview = async (ref: string) => {
   226→   226→    setPreviewLoading(true);
   227→   227→    try {
   228→   228→      const url = `/api/admin/qr-design/preview?reference=${ref}&sizeCm=${sizeCm}`;
   229→   229→      const res = await fetch(url);
   230→   230→      if (res.ok) {
   231→   231→        const blob = await res.blob();
   232→   232→        const objectUrl = URL.createObjectURL(blob);
   233→   233→        setPreviewUrl((prev) => {
   234→   234→          if (prev) URL.revokeObjectURL(prev);
   235→   235→          return objectUrl;
   236→   236→        });
   237→   237→      }
   238→   238→    } catch {
   239→   239→      // Silent fail for preview
   240→   240→    } finally {
   241→   241→      setPreviewLoading(false);
   242→   242→    }
   243→   243→  };
   244→   244→
   245→   245→  const downloadFile = async (url: string, filename: string) => {
   246→   246→    const res = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ references, sizeCm }) });
   247→   247→    if (!res.ok) {
   248→   248→      const err = await res.json().catch(() => ({ error: 'Erreur' }));
   249→   249→      setErrorMsg(err.error || "Erreur lors de l'export.");
   250→   250→      return;
   251→   251→    }
   252→   252→    const blob = await res.blob();
   253→   253→    const objectUrl = URL.createObjectURL(blob);
   254→   254→    const a = document.createElement('a');
   255→   255→    a.href = objectUrl;
   256→   256→    a.download = filename;
   257→   257→    document.body.appendChild(a);
   258→   258→    a.click();
   259→   259→    document.body.removeChild(a);
   260→   260→    URL.revokeObjectURL(objectUrl);
   261→   261→  };
   262→   262→
   263→   263→  const handleExportPdf = async () => {
   264→   264→    if (references.length === 0) return;
   265→   265→    setExportingPdf(true);
   266→   266→    setErrorMsg('');
   267→   267→    try {
   268→   268→      const date = new Date().toISOString().slice(0, 10);
   269→   269→      await downloadFile('/api/admin/qr-design/export-pdf', `QRTags-Design-${sizeCm}cm-${references.length}QR-${date}.pdf`);
   270→   270→      setSuccessMsg(`PDF exporté avec ${references.length} QR codes en grille A4 !`);
   271→   271→    } finally {
   272→   272→      setExportingPdf(false);
   273→   273→    }
   274→   274→  };
   275→   275→
   276→   276→  const handleExportZip = async () => {
   277→   277→    if (references.length === 0) return;
   278→   278→    setExportingZip(true);
   279→   279→    setErrorMsg('');
   280→   280→    try {
   281→   281→      const date = new Date().toISOString().slice(0, 10);
   282→   282→      await downloadFile('/api/admin/qr-design/export-zip', `QRTags-Design-${sizeCm}cm-${references.length}QR-${date}.zip`);
   283→   283→      setSuccessMsg(`ZIP exporté avec ${references.length} PNG haute résolution !`);
   284→   284→    } finally {
   285→   285→      setExportingZip(false);
   286→   286→    }
   287→   287→  };
   288→   288→
   289→   289→  const hasTemplate = templateInfo?.exists ?? false;
   290→   290→  const canExport = references.length > 0 && hasTemplate;
   291→   291→
   292→   292→  return (
   293→   293→    <div className="space-y-6">
   294→   294→      {/* ─── Step 1: Upload Template ─── */}
   295→   295→      <Card className="rounded-2xl">
   296→   296→        <CardHeader>
   297→   297→          <CardTitle className="flex items-center gap-2 text-slate-800 dark:text-white">
   298→   298→            <div className="w-8 h-8 rounded-full bg-gradient-to-r from-orange-500 to-purple-600 flex items-center justify-center text-white text-sm font-bold">1</div>
   299→   299→            Template de Design
   300→   300→          </CardTitle>
   301→   301→        </CardHeader>
   302→   302→        <CardContent>
   303→   303→          <div className="flex flex-col sm:flex-row items-start sm:items-center gap-4">
   304→   304→            {hasTemplate ? (
   305→   305→              <div className="flex items-center gap-3 flex-1">
   306→   306→                <div className="w-12 h-12 rounded-xl bg-emerald-50 dark:bg-emerald-900/20 flex items-center justify-center">
   307→   307→                  <CheckCircle className="w-6 h-6 text-emerald-600" />
   308→   308→                </div>
   309→   309→                <div>
   310→   310→                  <p className="font-medium text-slate-800 dark:text-white">Template configuré</p>
   311→   311→                  <p className="text-sm text-slate-500">
   312→   312→                    {templateInfo!.width}×{templateInfo!.height}px • {templateInfo!.format.toUpperCase()}
   313→   313→                    • {(templateInfo!.size / 1024).toFixed(0)} Ko
   314→   314→                  </p>
   315→   315→                </div>
   316→   316→                <Button variant="outline" size="sm" className="ml-auto" onClick={() => fileInputRef.current?.click()}>
   317→   317→                  <Upload className="w-4 h-4 mr-1" /> Remplacer
   318→   318→                </Button>
   319→   319→              </div>
   320→   320→            ) : (
   321→   321→              <div className="flex items-center gap-3 flex-1">
   322→   322→                <div className="w-12 h-12 rounded-xl bg-amber-50 dark:bg-amber-900/20 flex items-center justify-center">
   323→   323→                  <Image className="w-6 h-6 text-amber-600" />
   324→   324→                </div>
   325→   325→                <div>
   326→   326→                  <p className="font-medium text-slate-800 dark:text-white">Aucun template</p>
   327→   327→                  <p className="text-sm text-slate-500">Uploadez un design PNG (carré, fond transparent)</p>
   328→   328→                </div>
   329→   329→                <Button
   330→   330→                  className="ml-auto bg-gradient-to-r from-orange-500 to-purple-600 hover:from-orange-600 hover:to-purple-700 text-white"
   331→   331→                  onClick={() => fileInputRef.current?.click()}
   332→   332→                >
   333→   333→                  <Upload className="w-4 h-4 mr-1" /> Uploader le design
   334→   334→                </Button>
   335→   335→              </div>
   336→   336→            )}
   337→   337→            <input
   338→   338→              ref={fileInputRef}
   339→   339→              type="file"
   340→   340→              accept="image/png"
   341→   341→              className="hidden"
   342→   342→              onChange={handleUpload}
   343→   343→            />
   344→   344→          </div>
   345→   345→        </CardContent>
   346→   346→      </Card>
   347→   347→
   348→   348→      {/* ─── Step 2: Configure & Generate ─── */}
   349→   349→      <Card className="rounded-2xl">
   350→   350→        <CardHeader>
   351→   351→          <CardTitle className="flex items-center gap-2 text-slate-800 dark:text-white">
   352→   352→            <div className="w-8 h-8 rounded-full bg-gradient-to-r from-orange-500 to-purple-600 flex items-center justify-center text-white text-sm font-bold">2</div>
   353→   353→            Configuration
   354→   354→          </CardTitle>
   355→   355→        </CardHeader>
   356→   356→        <CardContent className="space-y-5">
   357→   357→          {/* Agency Selection */}
   358→   358→          <div className="space-y-2">
   359→   359→            <Label className="text-slate-700 dark:text-slate-300">Agence (optionnel)</Label>
   360→   360→            <Select value={agencyId} onValueChange={setAgencyId}>
   361→   361→              <SelectTrigger className="bg-white dark:bg-slate-800 border-slate-200 dark:border-slate-700">
   362→   362→                <SelectValue placeholder="Sans agence (stock général)" />
   363→   363→              </SelectTrigger>
   364→   364→              <SelectContent className="bg-white dark:bg-slate-900">
   365→   365→                <SelectItem value="">Sans agence (stock général)</SelectItem>
   366→   366→                {agencies.filter((a) => a.active).map((a) => (
   367→   367→                  <SelectItem key={a.id} value={a.id}>{a.name}</SelectItem>
   368→   368→                ))}
   369→   369→              </SelectContent>
   370→   370→            </Select>
   371→   371→          </div>
   372→   372→
   373→   373→          {/* Quantity */}
   374→   374→          <div className="space-y-2">
   375→   375→            <Label className="text-slate-700 dark:text-slate-300">
   376→   376→              Nombre de QR codes <span className="text-slate-400 font-normal">(max 100)</span>
   377→   377→            </Label>
   378→   378→            <div className="flex gap-2 items-center">
   379→   379→              <Input
   380→   380→                type="number" min={1} max={100} value={quantity}
   381→   381→                onChange={(e) => setQuantity(Math.min(100, Math.max(1, parseInt(e.target.value) || 1)))}
   382→   382→                className="w-28 bg-white dark:bg-slate-800 border-slate-200 dark:border-slate-700"
   383→   383→              />
   384→   384→              <div className="flex gap-1.5 flex-wrap">
   385→   385→                {QUICK_QTYS.map((q) => (
   386→   386→                  <button
   387→   387→                    key={q} onClick={() => setQuantity(q)}
   388→   388→                    className={cn(
   389→   389→                      'px-3 py-1.5 rounded-lg text-sm font-medium transition-all',
   390→   390→                      quantity === q
   391→   391→                        ? 'bg-gradient-to-r from-orange-500 to-purple-600 text-white shadow-sm'
   392→   392→                        : 'bg-slate-100 dark:bg-slate-800 text-slate-600 dark:text-slate-400 hover:bg-slate-200 dark:hover:bg-slate-700'
   393→   393→                    )}
   394→   394→                  >{q}</button>
   395→   395→                ))}
   396→   396→              </div>
   397→   397→            </div>
   398→   398→          </div>
   399→   399→
   400→   400→          {/* Size Selection */}
   401→   401→          <div className="space-y-2">
   402→   402→            <Label className="flex items-center gap-2 text-slate-700 dark:text-slate-300">
   403→   403→              <Ruler className="w-4 h-4" />
   404→   404→              Taille du sticker (cercle complet)
   405→   405→            </Label>
   406→   406→            <div className="grid grid-cols-3 gap-3">
   407→   407→              {SIZES.map((s) => (
   408→   408→                <button
   409→   409→                  key={s.value}
   410→   410→                  onClick={() => setSizeCm(s.value)}
   411→   411→                  className={cn(
   412→   412→                    'relative p-4 rounded-xl border-2 transition-all text-center',
   413→   413→                    sizeCm === s.value
   414→   414→                      ? 'border-orange-500 bg-orange-50 dark:bg-orange-900/20 shadow-md'
   415→   415→                      : 'border-slate-200 dark:border-slate-700 hover:border-slate-300 dark:hover:border-slate-600'
   416→   416→                  )}
   417→   417→                >
   418→   418→                  <div className={cn(
   419→   419→                    'text-2xl font-bold',
   420→   420→                    sizeCm === s.value ? 'text-orange-600' : 'text-slate-700 dark:text-slate-300'
   421→   421→                  )}>
   422→   422→                    {s.value} cm
   423→   423→                  </div>
   424→   424→                  <div className="text-xs text-slate-500 mt-1">{s.gridInfo}</div>
   425→   425→                  {sizeCm === s.value && (
   426→   426→                    <div className="absolute top-2 right-2 w-5 h-5 rounded-full bg-orange-500 flex items-center justify-center">
   427→   427→                      <CheckCircle className="w-3 h-3 text-white" />
   428→   428→                    </div>
   429→   429→                  )}
   430→   430→                </button>
   431→   431→              ))}
   432→   432→            </div>
   433→   433→          </div>
   434→   434→
   435→   435→          {/* Generate Button */}
   436→   436→          <Button
   437→   437→            className="w-full bg-gradient-to-r from-orange-500 to-purple-600 hover:from-orange-600 hover:to-purple-700 text-white rounded-xl h-12 text-base font-semibold shadow-lg"
   438→   438→            onClick={handleGenerate}
   439→   439→            disabled={generating || !hasTemplate}
   440→   440→          >
   441→   441→            {generating ? (
   442→   442→              <><Loader2 className="w-5 h-5 mr-2 animate-spin" /> Génération en cours...</>
   443→   443→            ) : (
   444→   444→              <><QrCode className="w-5 h-5 mr-2" /> Générer {quantity} QR code(s)</>
   445→   445→            )}
   446→   446→          </Button>
   447→   447→        </CardContent>
   448→   448→      </Card>
