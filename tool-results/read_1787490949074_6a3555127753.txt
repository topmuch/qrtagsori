   186→// ─── Image fallback strategy ───
   187→// Production DB products often have either:
   188→//   • image: null (seed not re-run)
   189→//   • image: '/images/shop/product-XXXX.jpeg' pointing to a file that does NOT exist
   190→//     (uploaded via admin panel, but the upload failed silently or was lost)
   191→//   • slug variants like 'pack-5-stickers-' (trailing dash) or 'packs-10-stickers'
   192→//     that don't match SHOP_PACKS_FALLBACK slugs exactly.
   193→//
   194→// To ALWAYS show a sachet image we:
   195→//   1. Map quantity → local pack image (most reliable, the sachet shows the right number)
   196→//   2. Map slug → local pack image (handles edge cases where quantity differs)
   197→//   3. On <img> onError, swap the broken src to the quantity-mapped local image
   198→//      (instead of hiding it and showing the number fallback, which is less pretty)
   199→const PACK_IMAGE_BY_QUANTITY: Record<number, string> = Object.fromEntries(
   200→  SHOP_PACKS_FALLBACK.map(p => [p.quantity, p.image])
   201→);
   202→const PACK_IMAGE_BY_SLUG: Record<string, string> = Object.fromEntries(
   203→  SHOP_PACKS_FALLBACK.map(p => [p.slug, p.image])
   204→);
   205→// Generic sachet used when quantity is unknown (e.g. Pack Revendeur qty=35)
   206→const GENERIC_PACK_IMAGE = '/images/shop/pack-15-stickers.png';
   207→
   208→/**
   209→ * Resolve the best local image for a given product.
   210→ *
   211→ * CRITICAL: we NEVER trust DB-uploaded images (pattern: /images/shop/product-*.jpeg)
   212→ * because they are almost always broken uploads (admin panel saved the path but
   213→ * the file was never written, or was lost during deploy). We only trust the 4
   214→ * known-good local sachet images that ship with the codebase.
   215→ *
   216→ * Strategy (in order):
   217→ *   1. If DB image matches the known-good local pattern → use it as-is.
   218→ *   2. Otherwise resolve by quantity (most reliable — sachet shows the right number).
   219→ *   3. Then by slug, slug-without-trailing-dash, slug-without-s.
   220→ *   4. Finally fall back to the generic pack-15 sachet.
   221→ *
   222→ * This guarantees the <img src> is ALWAYS a valid local path from the first render,
   223→ * so there is no flash of broken image, no 30s-then-disappear bug, no need to rely
   224→ * on the onError swap (which can race with React re-renders).
   225→ */
   226→function resolvePackImage(p: { image: string | null; quantity: number; slug: string }): string {
   227→  // Only trust DB image if it matches one of our 4 known-good local sachet paths
   228→  if (p.image && /^\/images\/shop\/pack-(3|5|10|15)-stickers\.png$/.test(p.image)) {
   229→    return p.image;
   230→  }
   231→  return (
   232→    PACK_IMAGE_BY_QUANTITY[p.quantity] ||
   233→    PACK_IMAGE_BY_SLUG[p.slug] ||
   234→    PACK_IMAGE_BY_SLUG[p.slug.replace(/[-\s]+$/, '')] || // strip trailing dash: 'pack-5-stickers-' → 'pack-5-stickers'
   235→    PACK_IMAGE_BY_SLUG[p.slug.replace(/^packs-/, 'pack-')] || // 'packs-10-stickers' → 'pack-10-stickers'
   236→    GENERIC_PACK_IMAGE
   237→  );
   238→}
   239→
   240→/**
   241→ * Decide what to use as the <img src>. Returns the DB image ONLY if it's safe
   242→ * (one of the 4 known-good local paths). Otherwise returns the fallback.
   243→ *
   244→ * This prevents the "30-second disappear" bug where:
   245→ *   1. Initial render uses SHOP_PACKS_FALLBACK (sachets show).
   246→ *   2. API responds ~30s later with broken DB image URLs.
   247→ *   3. <img src> switches to broken URL → broken image icon.
   248→ *   4. onError swap may race with React re-renders and lose the swap.
   249→ *
   250→ * By NEVER using a DB image that doesn't match the safe pattern, we eliminate
   251→ * the broken-image state entirely.
   252→ */
   253→function getSafeImageSrc(p: { image: string | null; fallbackImage: string }): string {
   254→  if (p.image && /^\/images\/shop\/pack-(3|5|10|15)-stickers\.png$/.test(p.image)) {
   255→    return p.image;
   256→  }
   257→  return p.fallbackImage;
   258→}
