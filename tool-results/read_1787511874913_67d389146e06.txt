   703→        open={authModalOpen}
   704→        onClose={() => { setAuthModalOpen(false); setPendingLinkRef(null); }}
   705→        defaultMode={authModalMode}
   706→        onSuccess={handleAuthSuccess}
   707→      />
   708→
   709→      {/* ─── Section Avis ─── */}
   710→      {isLoggedIn && !loading && displayBaggages.length > 0 && (
   711→        <div className="max-w-md mx-auto w-full px-4 pb-20">
   712→          {!showReviewForm ? (
   713→            <button
   714→              onClick={() => { setShowReviewForm(true); setReviewName(traveler?.name || ''); }}

   717→              <MessageSquare className="w-5 h-5 text-[#E3B23C] flex-shrink-0" />

   725→            <div className="rounded-2xl border-2 border-[#E3B23C]/40 bg-[#FFFDF5] overflow-hidden">

   753→                    >
   754→                      <option value="">— Aucun objet en particulier —</option>
   755→                      {displayBaggages.map(b => (
   756→                        <option key={b.reference} value={b.reference}>
   757→                          {b.objectName ? `${b.objectName} (${b.reference})` : b.reference}
   758→                        </option>
   759→                      ))}
   760→                    </select>
   761→                  </div>
   762→                )}
   763→                <div>
   764→                  <label className="block text-xs font-bold text-[#525252] mb-1.5">Votre note</label>
   765→                  <div className="flex items-center gap-1">
   766→                    {[1, 2, 3, 4, 5].map((star) => (
   767→                      <button
   768→                        key={star}
   769→                        type="button"
   770→                        onClick={() => setReviewRating(star)}
   771→                        className="transition-transform hover:scale-110"
   772→                      >
   773→                        <Star
   774→                          width={28}
   775→                          height={28}
   776→                          className={star <= reviewRating ? 'fill-current text-[#E3B23C]' : 'text-[#d4d4d4]'}
   777→                        />
   778→                      </button>
   779→                    ))}
   780→                    <span className="ml-2 text-sm font-bold text-[#1a1a1a]">{reviewRating}/5</span>
   781→                  </div>
   782→                </div>
   783→                <div>
   784→                  <label className="block text-xs font-bold text-[#525252] mb-1.5">Votre commentaire</label>
   785→                  <textarea
   786→                    value={reviewContent}
   787→                    onChange={(e) => setReviewContent(e.target.value)}
   788→                    placeholder="Partagez votre expérience..."
   789→                    rows={3}
   790→                    className="w-full px-4 py-2.5 rounded-xl bg-white border border-[#e5e5e5] text-[#1a1a1a] text-sm placeholder:text-[#a3a3a3] focus:ring-2 focus:ring-[#E3B23C] focus:border-transparent resize-none"
   791→                  />
   792→                </div>
   793→                {reviewError && (
   794→                  <p className="text-xs text-red-600 font-medium bg-red-50 rounded-lg p-2.5">{reviewError}</p>
   795→                )}
   796→                {reviewSuccess && (
   797→                  <p className="text-xs text-green-700 font-medium bg-green-50 rounded-lg p-2.5">✅ Merci pour votre avis ! Il sera publié après validation.</p>
   798→                )}
   799→                <button
   800→                  onClick={handleSubmitReview}
   801→                  disabled={reviewSubmitting || reviewContent.trim().length < 10 || reviewName.trim().length < 2}
   802→                  className="w-full py-3 bg-[#E3B23C] text-black text-sm font-bold rounded-xl hover:bg-[#E3B23C]/80 transition disabled:opacity-50 flex items-center justify-center gap-2"
   803→                >
   804→                  {reviewSubmitting ? (
   805→                    <Loader2 className="w-4 h-4 animate-spin" />
   806→                  ) : null}
   807→                  {reviewSubmitting ? 'Envoi en cours...' : 'Envoyer mon avis'}
   808→                </button>
   809→              </div>
   810→            </div>
   811→          )}
   812→        </div>
   813→      )}
   814→    </main>
   815→  );
   816→}

   818→// Helper: convert base64 VAPID key to Uint8Array
   819→function urlBase64ToUint8Array(base64String: string): Uint8Array {
   820→  const padding = '='.repeat((4 - base64String.length % 4) % 4);
   821→  const base64 = (base64String + padding).replace(/\-/g, '+').replace(/_/g, '/');
   822→  const rawData = window.atob(base64);
   823→  const outputArray = new Uint8Array(rawData.length);
   824→  for (let i = 0; i < rawData.length; ++i) {
   825→    outputArray[i] = rawData.charCodeAt(i);
   826→  }
   827→  return outputArray;
   828→}
